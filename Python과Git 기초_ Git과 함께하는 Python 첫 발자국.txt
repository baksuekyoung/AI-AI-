import json
import os

DATA_FILE = "prompts.json"
categories = ["텍스트 생성", "이미지 생성", "영상 생성", "페르소나", "자동화", "기타"]

# 데이터 불러오기 (보너스 1: JSON 영속화)

    if os.path.exists(DATA_FILE):
        try:

        except:
            pass
    
    # 파일이 없거나 오류 시 기본 작은도서관 데이터 3개 반환
    return [

            "category": "페르소나",
            "favorite": True,

            "category": "텍스트 생성",
            "favorite": False,

        },
        {
            "title": "마을 독서 모임 도서 추천 및 질문 생성",

            "category": "자동화",

# 데이터 저장하기
def save_data(prompts):
    with open(DATA_FILE, "w", encoding="utf-8") as f:
        json.dump(prompts, f, ensure_ascii=False, indent=4)

prompts = load_data()

def show_menu():
    """메인 메뉴 출력 및 선택 번호 반환"""
    print("\n=== 작은도서관 프롬프트 관리 시스템 ===")
    print("1. 프롬프트 추가")
    print("2. 프롬프트 목록")
    print("3. 카테고리별 조회")
    print("4. 프롬프트 검색")
    print("5. 프롬프트 상세 보기 (조회수 증가)")
    print("6. 즐겨찾기 관리")
    print("7. 즐겨찾기 목록")
    print("8. 프롬프트 수정 / 삭제 (보너스)")
    print("9. 인기 프롬프트 Top 목록 (보너스)")
    print("0. 종료")
    choice = input("선택: ").strip()
    return choice

def add_prompt():
    """새로운 프롬프트 추가"""
    print("\n=== 프롬프트 추가 ===")
    title = input("제목: ").strip()
    if not title:
        print("제목은 비워둘 수 없습니다.")
        return

    content = input("내용: ").strip()
    if not content:
        print("내용은 비워둘 수 없습니다.")
        return

    print("\n카테고리 선택:")
    for i, cat in enumerate(categories, 1):
        print(f"{i}) {cat}")
    
    cat_choice = input("선택: ").strip()
    if cat_choice.isdigit() and 1 <= int(cat_choice) <= len(categories):
        category = categories[int(cat_choice) - 1]
    else:
        print("올바른 번호를 선택하지 않아 '기타'로 지정됩니다.")
        category = "기타"

    new_prompt = {
        "title": title,
        "content": content,
        "category": category,
        "favorite": False,
        "views": 0
    }
    prompts.append(new_prompt)
    save_data(prompts)
    print("\n프롬프트가 추가되었습니다! (JSON 파일 저장 완료)")

def show_list():
    """저장된 모든 프롬프트 목록 출력"""
    print("\n=== 프롬프트 목록 ===")
    if not prompts:
        print("등록된 프롬프트가 없습니다.")
        return

    for i, p in enumerate(prompts, 1):
        fav = "⭐" if p["favorite"] else ""
        print(f"{i}. [{p['category']}] {p['title']} {fav} (조회수: {p.get('views', 0)})")
    
    print(f"\n총 {len(prompts)}개의 프롬프트")

def show_category():
    """카테고리별 프롬프트 조회"""
    print("\n=== 카테고리별 조회 ===")
    for i, cat in enumerate(categories, 1):
        print(f"{i}) {cat}")
    
    cat_choice = input("선택: ").strip()
    if not cat_choice.isdigit() or not (1 <= int(cat_choice) <= len(categories)):
        print("올바르지 않은 선택입니다.")
        return

    selected_cat = categories[int(cat_choice) - 1]
    print(f"\n[{selected_cat}] 카테고리 프롬프트:")
    
    matched = [p for p in prompts if p["category"] == selected_cat]
    if not matched:
        print("해당 카테고리에 프롬프트가 없습니다.")
        return

    for p in matched:
        fav = "⭐" if p["favorite"] else ""
        print(f"- {p['title']} {fav}")
    print(f"\n총 {len(matched)}개의 프롬프트")

def search_prompt():
    """키워드로 프롬프트 검색"""
    print("\n=== 프롬프트 검색 ===")
    keyword = input("검색어: ").strip()
    if not keyword:
        print("검색어를 입력해주세요.")
        return

    results = []
    for i, p in enumerate(prompts):
        if keyword in p["title"] or keyword in p["content"]:
            results.append((i + 1, p))

    print(f"\n검색 결과:")
    if not results:
        print("검색 결과가 없습니다.")
        return

    for idx, p in results:
        fav = "⭐" if p["favorite"] else ""
        print(f"{idx}. [{p['category']}] {p['title']} {fav}")
    
    print(f"\n{len(results)}개의 프롬프트를 찾았습니다.")

def show_detail():
    """프롬프트 상세 보기 및 조회수 증가 (보너스 2 반영)"""
    print("\n=== 프롬프트 상세 보기 ===")
    show_list()
    if not prompts:
        return

    num_input = input("번호 입력: ").strip()
    if not num_input.isdigit():
        print("숫자를 입력해주세요.")
        return

    idx = int(num_input) - 1
    if not (0 <= idx < len(prompts)):
        print("존재하지 않는 번호입니다.")
        return

    p = prompts[idx]
    p["views"] = p.get("views", 0) + 1  # 조회수 1 증가
    save_data(prompts)

    fav = "⭐" if p["favorite"] else "일반"
    print("\n" + "─" * 28)
    print(f"제목: {p['title']}")
    print(f"카테고리: {p['category']}")
    print(f"즐겨찾기: {fav}")
    print(f"조회수: {p['views']}")
    print("─" * 28)
    print(f"내용:\n{p['content']}")
    print("─" * 28)

def manage_favorite():
    """즐겨찾기 추가 및 해제"""
    print("\n=== 즐겨찾기 관리 ===")
    show_list()
    if not prompts:
        return

    num_input = input("프롬프트 번호 입력: ").strip()
    if not num_input.isdigit():
        print("숫자를 입력해주세요.")
        return

    idx = int(num_input) - 1
    if not (0 <= idx < len(prompts)):
        print("존재하지 않는 번호입니다.")
        return

    p = prompts[idx]
    p["favorite"] = not p["favorite"]
    save_data(prompts)
    
    status = "추가" if p["favorite"] else "해제"
    print(f"'{p['title']}' 프롬프트를 즐겨찾기에 {status}했습니다!")

def show_favorites():
    """즐겨찾기 목록 보기"""
    print("\n=== 즐겨찾기 목록 ===")
    favs = [p for p in prompts if p["favorite"]]
    if not favs:
        print("즐겨찾기된 프롬프트가 없습니다.")
        return

    for i, p in enumerate(favs, 1):
        print(f"{i}. [{p['category']}] {p['title']} ⭐")
    
    print(f"\n총 {len(favs)}개의 즐겨찾기")

def update_or_delete_prompt():
    """보너스 과제 2: 프롬프트 수정 및 삭제 기능"""
    print("\n=== 프롬프트 수정 / 삭제 ===")
    show_list()
    if not prompts:
        return

    num_input = input("작업할 프롬프트 번호 입력: ").strip()
    if not num_input.isdigit():
        print("숫자를 입력해주세요.")
        return

    idx = int(num_input) - 1
    if not (0 <= idx < len(prompts)):
        print("존재하지 않는 번호입니다.")
        return

    target = prompts[idx]
    print(f"\n선택된 프롬프트: {target['title']}")
    print("1. 수정하기")
    print("2. 삭제하기")
    sub_choice = input("선택: ").strip()

    if sub_choice == "1":
        new_title = input(f"새로운 제목 (기존: {target['title']}): ").strip()
        new_content = input(f"새로운 내용 (기존: {target['content']}): ").strip()
        
        if new_title:
            target["title"] = new_title
        if new_content:
            target["content"] = new_content
        
        save_data(prompts)
        print("프롬프트가 수정되었습니다!")

    elif sub_choice == "2":
        deleted = prompts.pop(idx)
        save_data(prompts)
        print(f"'{deleted['title']}' 프롬프트가 삭제되었습니다.")
    else:
        print("잘못된 입력입니다.")

def show_top_views():
    """보너스 과제 2: 조회수 기준 정렬(Top 목록) 기능"""
    print("\n=== 인기 프롬프트 Top 목록 (조회수 기준) ===")
    if not prompts:
        print("등록된 프롬프트가 없습니다.")
        return

    # 조회수가 높은 순으로 정렬
    sorted_prompts = sorted(prompts, key=lambda x: x.get("views", 0), reverse=True)

    for i, p in enumerate(sorted_prompts[:5], 1):  # 상위 5개 출력
        print(f"{i}. [{p['category']}] {p['title']} - 조회수: {p.get('views', 0)}회")

def main():
    """프로그램 메인 실행 루프"""
    while True:
        choice = show_menu()
        if choice == "1":
            add_prompt()
        elif choice == "2":
            show_list()
        elif choice == "3":
            show_category()
        elif choice == "4":
            search_prompt()
        elif choice == "5":
            show_detail()
        elif choice == "6":
            manage_favorite()
        elif choice == "7":
            show_favorites()
        elif choice == "8":
            update_or_delete_prompt()
        elif choice == "9":
            show_top_views()
        elif choice == "0":
            print("\n프로그램을 종료합니다. 이용해 주셔서 감사합니다!")
            break
        else:
            print("\n잘못된 번호입니다. 0~9 사이의 숫자를 입력해주세요.")

if __name__ == "__main__":
    main()
